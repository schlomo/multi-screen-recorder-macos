{{- /* A window screenshot from assets/images/, shown at the window's size in points and never wider than the page. Arguments: src, alt and caption. A file whose name ends in "-2x.png" (taken on a Retina screen) has 2 pixels per point and also gets a 1x version for screens that need no more. No "@2x": Cloudflare redirects a URL with "@" to its encoded form. */}} {{- $image := resources.Get .src }} {{- $scale := cond (strings.HasSuffix .src "-2x.png") 2 1 }} {{- $width := div $image.Width $scale }} {{- $height := div $image.Height $scale }} {{- $full := $image.Process "webp q90 text" }} {{- $shown := $full }} {{- $srcset := "" }} {{- if eq $scale 2 }} {{- $shown = $image.Process (printf "resize %dx webp q90 text" $width) }} {{- $srcset = printf "%s 1x, %s 2x" $shown.RelPermalink $full.RelPermalink }} {{- end }}
